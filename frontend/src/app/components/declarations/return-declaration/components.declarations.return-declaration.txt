import { Component, Input, OnInit } from '@angular/core';
import { CommonModule } from '@angular/common';
import { ReactiveFormsModule, FormBuilder, FormGroup, FormArray, Validators } from '@angular/forms';
import { NgbActiveModal } from '@ng-bootstrap/ng-bootstrap';
import { DeclarationService } from '../../../services/declaration.service';
import { MaterialService } from '../../../services/material.service';
import { ReturnDeclaration } from '../../../models/declaration';
import { Material } from '../../../models/material';

@Component({
  selector: 'app-return-declaration-form',
  standalone: true,
  imports: [CommonModule, ReactiveFormsModule],
  templateUrl: '../return-declaration-form/return-declaration-form.component.html',
  styleUrls: ['./return-declaration-form.component.css']
})
export class ReturnDeclarationFormComponent implements OnInit {
  @Input() declaration?: ReturnDeclaration;

  form!: FormGroup;
  submitted = false;
  isLoading = false;
  isEditMode = false;
  availableMaterials: Material[] = [];

  get materials(): FormArray {
    return this.form.get('materials') as FormArray;
  }

  get f() {
    return this.form.controls;
  }

  constructor(
    private fb: FormBuilder,
    public activeModal: NgbActiveModal,
    private declarationService: DeclarationService,
    private materialService: MaterialService
  ) {
    this.form = this.fb.group({
      declaredBy: ['', Validators.required],
      returnConditionNote: ['', Validators.required],
      note: [''],
      materials: this.fb.array([])
    });
  }

  ngOnInit(): void {
    this.loadAvailableMaterials();
    this.isEditMode = !!this.declaration;
    if (this.isEditMode && this.declaration) {
      this.form.patchValue(this.declaration);
    }
  }

  loadAvailableMaterials(): void {
    this.materialService.getMaterials().subscribe(materials => {
      this.availableMaterials = materials;
    });
  }

  addMaterial(): void {
    const group = this.fb.group({
      materialId: [null, Validators.required],
      quantity: [1, [Validators.required, Validators.min(1)]],
      condition: ['GOOD', Validators.required]
    });
    this.materials.push(group);
  }

  removeMaterial(index: number): void {
    this.materials.removeAt(index);
  }

  submit(): void {
    this.submitted = true;

    if (this.form.invalid || this.materials.length === 0) {
      return;
    }

    this.isLoading = true;
    const formValue = this.form.value;

    if (this.isEditMode && this.declaration) {
      this.declarationService.updateDeclaration(this.declaration.id, formValue).subscribe({
        next: (result) => {
          this.isLoading = false;
          this.activeModal.close(result);
        },
        error: (error) => {
          console.error('Error updating declaration:', error);
          this.isLoading = false;
        }
      });
    } else {
      this.declarationService.createReturnDeclaration(formValue).subscribe({
        next: (result) => {
          this.isLoading = false;
          this.activeModal.close(result);
        },
        error: (error) => {
          console.error('Error creating declaration:', error);
          this.isLoading = false;
        }
      });
    }
  }

  cancel(): void {
    this.activeModal.dismiss();
  }
}
